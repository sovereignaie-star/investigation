الملف 2: src/store/investigationStore.ts

```typescript
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

interface InvestigationStoreState {
  sessionId: string;
  nodes: any[];
  links: any[];
  lastSearches: Array<{ type: string; typeName: string; value: string; timestamp: number }>;
  isInvestigating: boolean;
  progress: number;
  totalNodes: number;
  fetchedNodes: number;
  errorNodes: number;
  maxDepth: number;
  startTime: number | null;
  apiKeys: Record<string, string>;
  currentGraph: any | null;

  setInvestigating: (value: boolean) => void;
  setProgress: (progress: number) => void;
  addNode: (node: any) => void;
  addLink: (link: any) => void;
  updateNode: (nodeId: string, updates: any) => void;
  removeNode: (nodeId: string) => void;
  updateStats: (stats: any) => void;
  setApiKey: (name: string, value: string) => void;
  removeApiKey: (name: string) => void;
  clearInvestigation: () => void;
  setGraph: (graph: any) => void;
}

export const useInvestigationStore = create<InvestigationStoreState>()(
  persist(
    (set, get) => ({
      sessionId: `inv-${Date.now()}`,
      nodes: [],
      links: [],
      lastSearches: [],
      isInvestigating: false,
      progress: 0,
      totalNodes: 0,
      fetchedNodes: 0,
      errorNodes: 0,
      maxDepth: 0,
      startTime: null,
      apiKeys: {},
      currentGraph: null,

      setInvestigating: (value) => set({ isInvestigating: value, startTime: value ? Date.now() : null }),
      setProgress: (progress) => set({ progress }),
      addNode: (node) => set((state) => ({ nodes: [...state.nodes, node] })),
      addLink: (link) => set((state) => ({ links: [...state.links, link] })),
      updateNode: (nodeId, updates) => set((state) => ({
        nodes: state.nodes.map((n) => n.id === nodeId ? { ...n, ...updates } : n),
      })),
      removeNode: (nodeId) => set((state) => ({
        nodes: state.nodes.filter((n) => n.id !== nodeId),
        links: state.links.filter((l) => l.sourceId !== nodeId && l.targetId !== nodeId),
      })),
      updateStats: (stats) => set({
        totalNodes: stats.totalNodes || 0,
        fetchedNodes: stats.fetchedNodes || 0,
        errorNodes: stats.errorNodes || 0,
        maxDepth: stats.maxDepth || 0,
      }),
      setApiKey: (name, value) => set((state) => ({
        apiKeys: { ...state.apiKeys, [name]: value },
      })),
      removeApiKey: (name) => set((state) => {
        const newKeys = { ...state.apiKeys };
        delete newKeys[name];
        return { apiKeys: newKeys };
      }),
      clearInvestigation: () => set({
        sessionId: `inv-${Date.now()}`,
        nodes: [],
        links: [],
        isInvestigating: false,
        progress: 0,
        totalNodes: 0,
        fetchedNodes: 0,
        errorNodes: 0,
        maxDepth: 0,
        startTime: null,
        currentGraph: null,
      }),
      setGraph: (graph) => set({ currentGraph: graph }),
    }),
    {
      name: 'investigation-room-storage',
      partialize: (state) => ({
        apiKeys: state.apiKeys,
        lastSearches: state.lastSearches,
      }),
    }
  )
);
```
